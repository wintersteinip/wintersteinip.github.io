function Assert-Admin {
    $id = [Security.Principal.WindowsIdentity]::GetCurrent()
    $p  = New-Object Security.Principal.WindowsPrincipal($id)
    if (-not $p.IsInRole([System.Security.Principal.WindowsBuiltInRole]::Administrator)) {
        throw "Please run this script as Administrator."
    }
}

function New-SessionLogRoot {
    param(
        [Parameter(Mandatory=$true)][string]$BaseDestRoot,
        [Parameter(Mandatory=$true)][string]$Prefix
    )

    if (-not (Test-Path $BaseDestRoot)) { throw "Destination base root not found: $BaseDestRoot" }

    for ($i=1; $i -le 9999; $i++) {
        $suffix = "{0:0000}" -f $i
        $dir = Join-Path $BaseDestRoot ("${Prefix}.$suffix")
        if (-not (Test-Path $dir)) {
            New-Item -ItemType Directory -Path $dir -Force | Out-Null

            $stampOld = (Get-Date).ToUniversalTime().ToString("yyyyMMddHHmmssfffffff").Trim()
            $logFile = Join-Path $dir ("OLD.$stampOld.TXT")

            New-Item -Type File -Path $logFile -Force | Out-Null

            return [pscustomobject]@{
                SessionDir = $dir
                LogFile    = $logFile
                SessionId  = $suffix
                StampOld   = $stampOld
            }
        }
    }

    throw "Could not create new session log root: all ${Prefix}.0001..${Prefix}.9999 exist."
}

function Append-LogLine {
    param(
        [Parameter(Mandatory=$true)][string]$LogFile,
        [Parameter(Mandatory=$true)][string]$Line
    )
    $Line = ($Line -replace "`r","" -replace "`n","")
    Add-Content -Path $LogFile -Value $Line -Encoding UTF8
}

function Get-NextMergeRoot {
    param(
        [Parameter(Mandatory=$true)][string]$BasePath,
        [Parameter(Mandatory=$true)][string]$FileRelPath
    )

    if (-not (Test-Path $BasePath)) {
        New-Item -ItemType Directory -Path $BasePath -Force | Out-Null
    }

    $candidate = $BasePath
    $destFile  = Join-Path $candidate $FileRelPath
    if (-not (Test-Path $destFile)) { return $candidate }

    for ($i=1; $i -le 9999; $i++) {
        $suffix = "{0:0000}" -f $i
        $cand   = "${BasePath}.$suffix"
        $destFile = Join-Path $cand $FileRelPath
        if (-not (Test-Path $destFile)) {
            if (-not (Test-Path $cand)) {
                New-Item -ItemType Directory -Path $cand -Force | Out-Null
            }
            return $cand
        }
    }

    throw "Collision routing failed: all F:\@MERGE.0001..@MERGE.9999 already contain $FileRelPath"
}

function TakeOwnership-And-GrantAdmins {
    param(
        [Parameter(Mandatory=$true)][string]$Path
    )
    $quoted = $Path.Replace('"','')

    cmd /c "takeown /F `"$quoted`" /R /D Y" | Out-Null
    cmd /c "icacls `"$quoted`" /inheritance:e /grant:r Administrators:(F) /T /C" | Out-Null
}

function New-DirIfMissing {
    param([Parameter(Mandatory=$true)][string]$Path)
    if (-not (Test-Path $Path)) {
        New-Item -ItemType Directory -Path $Path -Force | Out-Null
    }
}

function Move-WithFileLevelCollisionRouting {
    param(
        [Parameter(Mandatory=$true)][string]$SourceRoot,
        [Parameter(Mandatory=$true)][string]$DestBaseRoot,
        [Parameter(Mandatory=$true)][string]$LogFile
    )

    New-DirIfMissing $DestBaseRoot

    $files = Get-ChildItem -LiteralPath $SourceRoot -Recurse -File -Force
    foreach ($f in $files) {
        $rel = $f.FullName.Substring($SourceRoot.Length).TrimStart('\')
        $chosenRoot = Get-NextMergeRoot -BasePath $DestBaseRoot -FileRelPath $rel

        $destPath = Join-Path $chosenRoot $rel
        $destDir  = Split-Path -Parent $destPath
        New-DirIfMissing $destDir

        $ts = (Get-Date).ToString("yyyy-MM-ddTHH:mm:ss.fffffff")
        Append-LogLine -LogFile $LogFile -Line ("MOVE_OLDPATH={0}|NEW_PATH={1}|UTC={2}" -f $f.FullName, $destPath, $ts)

        Move-Item -LiteralPath $f.FullName -Destination $destPath -Force
    }

    Get-ChildItem -LiteralPath $SourceRoot -Recurse -Directory -Force |
        Sort-Object FullName -Descending |
        ForEach-Object {
            try {
                if ((Get-ChildItem -LiteralPath $_.FullName -Force | Measure-Object).Count -eq 0) {
                    Remove-Item -LiteralPath $_.FullName -Force -ErrorAction SilentlyContinue
                }
            } catch {}
        }
}

function Rename-FolderToMergeInPlace {
    param([Parameter(Mandatory=$true)][string]$SelectedFolder)

    $parent      = Split-Path -Parent $SelectedFolder
    $currentName = Split-Path -Leaf $SelectedFolder
    $target      = Join-Path $parent "@MERGE"

    if ($currentName -ieq "@MERGE") { return $SelectedFolder }

    if (Test-Path $target) {
        throw "Cannot rename in-place: $target already exists. Please choose a different folder or move it out of the way."
    }

    Rename-Item -LiteralPath $SelectedFolder -NewName "@MERGE"
    return $target
}

function Restore-FolderLevel {
    param([Parameter(Mandatory=$true)][string]$SelectedFolder)

    $parent = Split-Path -Parent $SelectedFolder
    $target = Join-Path $parent "@MERGE"

    if (Test-Path $target) {
        throw "Target already exists: $target"
    }

    Rename-Item -LiteralPath $SelectedFolder -NewName "@MERGE" -ErrorAction Stop
    TakeOwnership-And-GrantAdmins -Path $target
}

function MediaStein-C {
    param(
        [Parameter(Mandatory=$true)][string]$SourceRoot,
        [Parameter(Mandatory=$true)][string]$DestBaseRoot,
        [Parameter(Mandatory=$true)][string]$LogFile
    )

    New-DirIfMissing -Path $DestBaseRoot

    $files = Get-ChildItem -LiteralPath $SourceRoot -Recurse -File -Force
    foreach ($f in $files) {
        $rel = $f.FullName.Substring($SourceRoot.Length).TrimStart('\')

        $dt = $f.LastWriteTime
        $year  = $dt.ToString("yyyy")
        $month = $dt.ToString("MM")
        $day   = $dt.ToString("dd")

        $relDir = Split-Path -Parent $rel
        $leaf   = Split-Path -Leaf $rel

        $ext        = [System.IO.Path]::GetExtension($leaf)
        $baseNoExt  = [System.IO.Path]::GetFileNameWithoutExtension($leaf)

        $stamp    = $dt.ToString("yyyyMMdd_HHmmss_ffff")
        $newLeaf  = "{0}_{1}{2}" -f $stamp, $baseNoExt, $ext

        $destPath = Join-Path $DestBaseRoot $year
        $destPath = Join-Path $destPath $month
        $destPath = Join-Path $destPath $day

        if ($relDir -and $relDir -ne '.') {
            $destPath = Join-Path $destPath $relDir
        }

        $destPath = Join-Path $destPath $newLeaf
        $destDir  = Split-Path -Parent $destPath
        New-DirIfMissing -Path $destDir

        $ts = (Get-Date).ToString("yyyy-MM-ddTHH:mm:ss.fffffff")
        Append-LogLine -LogFile $LogFile -Line ("MEDIASTEIN_C_MOVE_OLD={0}|NEW={1}|UTC={2}" -f $f.FullName, $destPath, $ts)

        try { TakeOwnership-And-GrantAdmins -Path $destDir } catch {}
        Move-Item -LiteralPath $f.FullName -Destination $destPath -Force
    }

    Get-ChildItem -LiteralPath $SourceRoot -Recurse -Directory -Force |
        Sort-Object FullName -Descending |
        ForEach-Object {
            try {
                if ((Get-ChildItem -LiteralPath $_.FullName -Force | Measure-Object).Count -eq 0) {
                    Remove-Item -LiteralPath $_.FullName -Force -ErrorAction SilentlyContinue
                }
            } catch {}
        }
}

function Refresh-LogBox {
    if (Test-Path $session.LogFile) {
        try {
            $txtLog.Text = (Get-Content -Path $session.LogFile -Encoding UTF8 | Select-Object -Last 500) -join "`r`n"
            $txtLog.SelectionStart = $txtLog.TextLength
            $txtLog.ScrollToCaret()
        } catch {}
    }
}

$timer = New-Object System.Windows.Forms.Timer
$timer.Interval = 1000
$timer.add_Tick({ Refresh-LogBox })
$timer.Start()

$btnMove.Add_Click({
    try {
        $folderDialog.Description = "Select the folder to MOVE (it will be renamed to @MERGE in-place)."
        $res = $folderDialog.ShowDialog()
        if ($res -ne [System.Windows.Forms.DialogResult]::OK) { return }

        Append-LogLine -LogFile $session.LogFile -Line ("MOVE_SESSION_REQUESTED|Selected={0}|UTC={1}" -f $folderDialog.SelectedPath, (Get-Date).ToUniversalTime().ToString("o"))

        $renamedRoot = Rename-FolderToMergeInPlace -SelectedFolder $folderDialog.SelectedPath
        Append-LogLine -LogFile $session.LogFile -Line ("RENAME_ROOT|OldRoot={0}|NewRoot={1}|UTC={2}" -f $folderDialog.SelectedPath, $renamedRoot, (Get-Date).ToUniversalTime().ToString("o"))

        TakeOwnership-And-GrantAdmins -Path $renamedRoot
        Move-WithFileLevelCollisionRouting -SourceRoot $renamedRoot -DestBaseRoot $destBase -LogFile $session.LogFile

        if (Test-Path $destBase) { TakeOwnership-And-GrantAdmins -Path $destBase }

        $parentOfBase = Split-Path -Parent $destBase
        $suffixRoots = Get-ChildItem -LiteralPath $parentOfBase -Directory -Force |
            Where-Object { $_.Name -match '^@MERGE\.\d{4}$' }

        foreach ($r in $suffixRoots) {
            if (Test-Path $r.FullName) { TakeOwnership-And-GrantAdmins -Path $r.FullName }
        }

        Append-LogLine -LogFile $session.LogFile -Line ("MOVE_SESSION_COMPLETE|UTC={0}" -f (Get-Date).ToUniversalTime().ToString("o"))

        [System.Windows.Forms.MessageBox]::Show(
            "Move/Merge complete.`nLog: $($session.LogFile)",
            "Done",
            [System.Windows.Forms.MessageBoxButtons]::OK,
            [System.Windows.Forms.MessageBoxIcon]::Information
        ) | Out-Null
    } catch {
        Append-LogLine -LogFile $session.LogFile -Line ("MOVE_SESSION_FAILED|UTC=$((Get-Date).ToUniversalTime().ToString('o'))|ERR=$($_.Exception.Message)")
        [System.Windows.Forms.MessageBox]::Show($_.Exception.Message, "Error", 'OK', 'Error') | Out-Null
    } finally {
        Refresh-LogBox
    }
})

$btnRestore.Add_Click({

**One debugging tip:** if “TopMost” ever seems to stop working while you’re interacting with dialogs, your move/restore commands may be blocking the UI thread. The log timer won’t tick and Windows may visually “focus” other windows—so the next step would be running the heavy move/restore work in a background thread while UI stays responsive.
